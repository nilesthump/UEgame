cmake_minimum_required(VERSION 3.28)
project(ExperimentCore LANGUAGES CXX)
if(NOT CMAKE_VERSION VERSION_LESS "3.29")
  message(FATAL_ERROR "Reference checks require CMake 3.28.x")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64"
   OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION MATCHES "^13\\.")
  message(FATAL_ERROR "Reference checks require Linux x86_64 GCC 13")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_UNITY_BUILD OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CORE_ROOT Source/UEgame/Private/ExperimentCore)
set(CORE_SOURCES ${CORE_ROOT}/src/Session/ExperimentSession.cpp)
# UBT discovers all module cpp files: reject omissions in this explicit list.
file(GLOB_RECURSE DISCOVERED_CORE CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} ${CORE_ROOT}/src/*.cpp)
list(SORT CORE_SOURCES)
list(SORT DISCOVERED_CORE)
if(NOT "${CORE_SOURCES}" STREQUAL "${DISCOVERED_CORE}")
  message(FATAL_ERROR "CMake/UBT core source sets differ")
endif()
add_library(experiment_core STATIC ${CORE_SOURCES})
target_include_directories(experiment_core PUBLIC ${CORE_ROOT}/include)
target_compile_options(experiment_core PUBLIC -fno-exceptions -fno-rtti -pedantic-errors -Wall -Wextra -Werror)
include(CTest)
if(NOT BUILD_TESTING)
  message(FATAL_ERROR "Reference validation requires BUILD_TESTING=ON")
endif()
add_executable(experiment_core_checks Tests/ExperimentCore/main.cpp)
target_include_directories(experiment_core_checks PRIVATE Tests/Shared)
target_link_libraries(experiment_core_checks PRIVATE experiment_core)
add_test(NAME B01.Session.WorldReady COMMAND experiment_core_checks B01.Session.WorldReady)
add_test(NAME B01.Session.IndependentStarts COMMAND experiment_core_checks B01.Session.IndependentStarts)
add_test(NAME B01.Session.FactIdentityAndDelivery COMMAND experiment_core_checks B01.Session.FactIdentityAndDelivery)
add_test(NAME B01.Session.StableBatchDeathPriority COMMAND experiment_core_checks B01.Session.StableBatchDeathPriority)
add_test(NAME B01.Session.AftermathAndMissingMethod COMMAND experiment_core_checks B01.Session.AftermathAndMissingMethod)
add_test(NAME B01.Session.EffectEligibility COMMAND experiment_core_checks B01.Session.EffectEligibility)
set_tests_properties(B01.Session.WorldReady B01.Session.IndependentStarts B01.Session.FactIdentityAndDelivery
  B01.Session.StableBatchDeathPriority B01.Session.AftermathAndMissingMethod B01.Session.EffectEligibility
  PROPERTIES TIMEOUT 10)
file(GLOB_RECURSE CORE_HEADERS CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/${CORE_ROOT}/include ${CORE_ROOT}/include/*.h)
foreach(HEADER IN LISTS CORE_HEADERS)
  string(MAKE_C_IDENTIFIER "${HEADER}" HEADER_ID)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/${HEADER_ID}.cpp "#include \"${HEADER}\"\n")
  add_library(header_${HEADER_ID} OBJECT ${CMAKE_CURRENT_BINARY_DIR}/${HEADER_ID}.cpp)
  target_link_libraries(header_${HEADER_ID} PRIVATE experiment_core)
endforeach()
